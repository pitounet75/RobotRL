# Anticogging steps input_pos each bin; you must reach that setpoint immediately.
# INPUT_MODE 5 = INPUT_MODE_TRAP_TRAJ — planner lags input_pos, index often stalls at 1+.
# INPUT_MODE 1 = INPUT_MODE_PASSTHROUGH — required here (set even if you use trap for normal moves).
odrv0.axis0.controller.config.control_mode = CONTROL_MODE_POSITION_CONTROL
odrv0.axis0.controller.config.input_mode = INPUT_MODE_PASSTHROUGH

# --- Anticogging: reset stored map, run calibration ---
odrv0.axis0.controller.anticogging_valid = False
odrv0.axis0.controller.config.anticogging.anticogging_enabled = False
odrv0.axis0.controller.config.anticogging.pre_calibrated = False

# --- Loosen the "settled" band (almost always required vs stock-like defaults) ---
# Firmware declares a bin done only when |pos err| and |vel| stay inside these limits for many
# control cycles. Defaults such as calib_pos_threshold=500 mean |err| <= 500 encoder counts;
# if pos_estimate hunts (limit cycle), errors are often ~0.005 turn (~10k+ counts) so index
# never advances. Same idea for velocity spikes vs calib_vel_threshold.
# Widen first so the full forward sweep (3600 bins) can finish; tighten thresholds later for a cleaner map if you want.
odrv0.axis0.controller.config.anticogging.calib_pos_threshold = 250.0
odrv0.axis0.controller.config.anticogging.calib_vel_threshold = 0.1 * float(odrv0.axis0.encoder.config.cpr)
odrv0.axis0.controller.input_pos=0
